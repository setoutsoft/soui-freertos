# =============================================================================
# soui -- swinx FreeRTOS platform layer + utilities + SOUI core + render-gdi
# as ONE ESP-IDF component.
#
# Source lists / defines / includes are a transcription of
#   <soui-freertos>/cmake/SouiFreeRTOS.cmake:430-477 (soui_fr_add_swinx)
#   and :478+ (soui_fr_add_soui) -- i.e. exactly the set that linked with 0 undefined
#   symbols on the ARM-newlib target (see soui_fr/GAP-REPORT.md).
#
# Three ESP32-S3 specific decisions:
#  1. The vendored FreeRTOS kernel is NOT compiled.  ESP-IDF supplies its own
#     SMP kernel (REQUIRES freertos); SOUI_FREERTOS_REAL makes
#     swinx_freertos_api.h forward to the IDF task/semaphore API.
#  2. framebuffer.cpp is EXCLUDED.  Its static 320x240 ARGB32 array is 300KB
#     of .bss, which would land in the ESP32-S3's scarce internal SRAM.
#     framebuffer_idf.cpp implements the identical C API backed by
#     heap_caps_malloc(MALLOC_CAP_SPIRAM).
#  3. The STL shim directory stl/ is deliberately NOT on the include path:
#     it contains headers literally named "mutex", "thread",
#     "condition_variable", ... which would shadow IDF's libstdc++ ones.
#     winobjs.cpp and the split platform TUs still #include
#     "swinx_freertos_api.h", so a forwarding header is generated in GEN_DIR
#     instead.
# =============================================================================

# IDF evaluates a component's CMakeLists TWICE: once in a standalone
# "component_get_requirements" cmake process (where variables set by the
# top-level project are NOT visible) and once in the real build.  Deriving
# SOUI_ROOT from this file's own location keeps both passes consistent.
#   <soui4>/soui-freertos/esp32s3/components/soui/../../../.. == <soui4>
get_filename_component(SOUI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT EXISTS "${SOUI_ROOT}/SOUI/include/souistd.h")
    message(FATAL_ERROR "[soui] SOUI_ROOT 推导失败: ${SOUI_ROOT}")
endif()

set(SWINX_DIR "${SOUI_ROOT}/swinx")
set(SOUI_DIR  "${SOUI_ROOT}/SOUI")
set(UTIL_DIR  "${SOUI_ROOT}/utilities")
set(GEN_DIR   "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${GEN_DIR}")

# ---- FreeRTOS kernel include dirs -----------------------------------------
# Since IDF v5.3 the kernel headers live in FreeRTOS-Kernel/include/freertos/
# and IDF's own code says #include "freertos/FreeRTOS.h".  swinx (written
# against a vanilla kernel tree) says #include "FreeRTOS.h", so the inner
# directory has to be added explicitly -- `REQUIRES freertos` only exports
# FreeRTOS-Kernel/include.
set(_idf_root "$ENV{IDF_PATH}")
if(NOT _idf_root AND IDF_PATH)
    set(_idf_root "${IDF_PATH}")
endif()
# IDF serialises INCLUDE_DIRS into component_requires.cmake as C strings, so a
# Windows path with backslashes turns "\w" into an invalid escape and cmake
# stops with "Invalid character escape '\w'".  Always store forward slashes.
file(TO_CMAKE_PATH "${_idf_root}" _idf_root)
set(FREERTOS_KERNEL_INC "${_idf_root}/components/freertos/FreeRTOS-Kernel/include/freertos")
if(NOT EXISTS "${FREERTOS_KERNEL_INC}/FreeRTOS.h")
    message(FATAL_ERROR
        "[soui] 找不到 FreeRTOS 内核头文件目录: ${FREERTOS_KERNEL_INC}\n"
        "请在已激活 ESP-IDF 的 shell 中构建（. .\\idf-env.ps1）。")
endif()

# ---- SOUI's config.h (separate from soui_cairo/config.h) -------------------
file(WRITE "${GEN_DIR}/config.h" "\
/* config.h -- generated by soui/CMakeLists.txt: static-lib SOUI build */
#define LIB_SOUI_COM
#define LIB_CORE
#define DISABLE_SWNDSPY 1
")

# ---- forwarding headers ----------------------------------------------------
# These let main/ and the swinx TUs reach the FreeRTOS platform headers
# without exporting swinx/src/platform/freertos/ wholesale (that directory
# also carries dirent.h / dlfcn.h shims that would shadow IDF's newlib ones).
file(WRITE "${GEN_DIR}/swinx_freertos_api.h"
     "#pragma once\n#include \"${SWINX_DIR}/src/platform/freertos/stl/swinx_freertos_api.h\"\n")
file(WRITE "${GEN_DIR}/framebuffer.h"
     "#pragma once\n#include \"${SWINX_DIR}/src/platform/freertos/framebuffer.h\"\n")
file(WRITE "${GEN_DIR}/input.h"
     "#pragma once\n#include \"${SWINX_DIR}/src/platform/freertos/input.h\"\n")

# ---- imgdecoder-stb + aupng -------------------------------------------------
# <img skin="app_logo"> / SImageWnd::SetImage() go through
# IRenderFactory::GetImgDecoder(), which is NULL until a factory is installed
# (SAppCfg.h:476 does exactly that on the desktop).  imgdecoder-stb is the
# dependency-free option: stb_image for PNG/JPG/GIF, aupng for APNG.
#
# This component is compiled with -w because both are third-party C/C++ and
# trip IDF's -Werror=all the same way pixman does.
set(IMGDEC_STB_DIR "${SOUI_ROOT}/components/imgdecoder-stb")

# ---- <edit>: lightweight SEditLite, richedit41 is NOT compiled --------------
# <edit> used to be backed by the vendored richedit41 + msls31 here: ~250
# translation units of 2001-era Win32 C++ that pushed the app image past 4MB
# (hence the 6MB partition) and cost most of the 15~20 minute clean build.
# SOUI now ships a self-contained plain-text implementation (SEditLite.cpp,
# selected with SOUI_EDIT_LITE -- see SOUI/include/control/SEdit.h) with the
# same XML attribute surface and the same IEdit interface, so the control was
# switched over and the engine unlinked.  Rich text (RTF, char-level
# formatting) is the only thing lost; this board has no keyboard and no IME to
# edit it with anyway.
#
# Remove SOUI_EDIT_LITE from target_compile_definitions below (and re-add the
# richedit41 source/include block from git history) to go back.

idf_component_register(
    SRCS
        ${SWINX_CORE_SRCS} ${SWINX_CMNCTL_SRCS}
        ${SWINX_PLATFORM_SRCS} ${SWINX_COMMON_SRCS} ${SWINX_GDI_SRCS}
        ${UTIL_SRCS} ${RENDERGDI_SRCS} ${IMGDEC_SRCS}
        "${SOUI_DIR}/src/SApp.cpp"
        "${SOUI_DIR}/src/SouiFactory.cpp"
        ${SOUI_CORE_SRCS} ${SOUI_RESMGR_SRCS} ${SOUI_HELPER_SRCS}
        ${SOUI_LAYOUT_SRCS} ${SOUI_MATRIX_SRCS} ${SOUI_EVENT_SRCS}
        ${SOUI_ANIM_SRCS} ${SOUI_VANIM_SRCS} ${SOUI_CTRL_SRCS}
        "${CMAKE_CURRENT_LIST_DIR}/framebuffer_idf.cpp"
    INCLUDE_DIRS
        "${GEN_DIR}"
        "${CMAKE_CURRENT_LIST_DIR}/include"
        "${SWINX_DIR}"
        "${SWINX_DIR}/include"
        "${SWINX_DIR}/src"
        "${SWINX_DIR}/src/gdi/cairo"
        "${SOUI_ROOT}"
        "${SOUI_ROOT}/components"
        "${SOUI_ROOT}/components/render-gdi"
        "${SOUI_ROOT}/components/render-gdi/linux"
        "${SOUI_ROOT}/third-part/nanosvg"
        "${SOUI_DIR}"
        "${SOUI_DIR}/include"
        "${SOUI_DIR}/src"
        "${UTIL_DIR}/include"
    PRIV_INCLUDE_DIRS
        "${GEN_DIR}"
        "${FREERTOS_KERNEL_INC}"
        "${SWINX_DIR}/src/platform/freertos"
        "${SWINX_DIR}/src/platform/freertos/uuid"
        "${IMGDEC_STB_DIR}"
        "${STB_DIR}"
        "${AUPNG_DIR}/include"
        "${AUPNG_DIR}/src"
    REQUIRES
        soui_cairo
        freertos
        heap
        pthread
        # esp_timer_get_time(): microsecond clock used by the temporary [prof]
        # instrumentation in SOUI/src/core/SHostWnd.cpp.
        esp_timer
    # static COM registrations (SOUI controls, render factory, skins) live in
    # objects no caller references by name; without --whole-archive the linker
    # drops them and SApplication cannot create anything.
    WHOLE_ARCHIVE
)

target_compile_definitions(${COMPONENT_LIB} PUBLIC
    SOUI_PLATFORM_FREERTOS
    SOUI_FREERTOS_REAL
    SOUI_PLATFORM_ESPIDF
    SOUI_ENABLE_SVG
    # <edit> = SEditLite (SOUI/src/control/SEditLite.cpp) instead of the
    # RichEdit-backed SRichEdit.  PUBLIC is mandatory: the macro changes the
    # class definition of SEdit, so main/ (which includes SouiCtrls.h) has to
    # see the same value or the two sides disagree on the vtable layout.
    SOUI_EDIT_LITE
    ENABLE_THUNK=0
    # swinx/src/uniconv.h: derive wchar_t size from the toolchain (4 on
    # Xtensa newlib), never from a hand-written constant.
    WCHAR_SIZE=__SIZEOF_WCHAR_T__
    # NOTE: ESP-IDF's xTaskCreate() stack depth is measured in BYTES, not
    # FreeRTOS "words" (the macro name is historical).
    #
    # 11.10 correction: raising this from 16KB to 32KB did NOT fix the
    # "A stack overflow in task soui_ui" panic -- it only looked like it,
    # because 32KB happened to survive the run being watched.  This macro is
    # the default stack for helper tasks swinx creates internally; the UI task
    # gets its stack from kUiTaskStackBytes in main/app_main.cpp, and that is
    # the one the paint recursion overflows.  Kept at 32KB (ample for the helper
    # tasks, and the old 16KB was genuinely tight for them).
    SWINX_FR_DEFAULT_STACK_WORDS=32768)
target_compile_definitions(${COMPONENT_LIB} PRIVATE
    SOUI_EXPORTS
    # cursoricon.cpp (LoadImageW/A, CreateIconFromResource) is a POSIX-platform
    # file reused as-is: it has no mmap on bare metal, and on a Windows-host
    # build of this shim the local definitions must win over user32's imports.
    SWINX_NO_MMAP
    SWINX_OWN_USER32)

target_compile_options(${COMPONENT_LIB} PRIVATE
    -w
    # NOT -Os: this option list comes after CMAKE_CXX_FLAGS, so a -Os here
    # silently overrides the project-wide -O2 and the whole SOUI/cairo/pixman
    # tree was being built for size while the rest of the firmware used -O2.
    # Everything in this component is hot rasteriser code; the 6MB app
    # partition has room for it.
    -O2
    -fexceptions
    # IDEA/IDF default is -std=gnu++2b (C++23).  SOUI/swinx are a C++14-era
    # codebase and one construct is outright broken by C++20: msgbox.cpp does
    #   list<ButtonInfo>.push_back({ IDOK, u8"确定" });
    # In C++20 u8"" is const char8_t[] and no longer converts to std::string,
    # so all 17 brace-initialised push_back calls stop compiling.  gnu++17 is
    # also what the host and Cortex-M3 builds of this port use.
    -std=gnu++17
    "$<$<COMPILE_LANGUAGE:CXX>:-include>"
    "$<$<COMPILE_LANGUAGE:CXX>:soui_idf_compat.h>")

# SViewBase.cpp includes neither souistd.h nor a header defining CRect before
# use.  Using a second "-include" option would be deduplicated by CMake and
# turn its argument into a stray input file (see SouiFreeRTOS.cmake:466-467),
# so gate the include from inside soui_idf_compat.h with a source-level
# COMPILE_DEFINITIONS flag instead.
set_source_files_properties("${SOUI_DIR}/src/control/SViewBase.cpp" PROPERTIES
    COMPILE_DEFINITIONS "SOUI_FR_INCLUDE_SOUISTD=1")
