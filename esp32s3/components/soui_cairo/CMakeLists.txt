# =============================================================================
# soui_cairo -- vendored cairo 1.18.4 / pixman 0.46.5 / libpng / zlib as an
# ESP-IDF component (pure C).
#
# Source lists, excludes and generated headers are a transcription of
#   <soui-freertos>/cmake/SouiFreeRTOS.cmake:285-429 (soui_fr_add_thirdparty)
#   and <soui-freertos>/soui_fr/CMakeLists.txt:302-394.
#
# WHY A SEPARATE COMPONENT: cairo's cairoint.h:50 does an unconditional
#   #include "config.h"
# while SOUI's utilities-def.h includes a DIFFERENT config.h (LIB_CORE /
# LIB_SOUI_COM).  Two same-named files cannot share one include directory
# (SouiFreeRTOS.cmake:70-73 documents this).  IDF components each own a
# private include path, so splitting at the C/C++ boundary solves it cleanly.
# =============================================================================

# IDF evaluates a component's CMakeLists TWICE: once in a standalone
# "component_get_requirements" cmake process (where variables set by the
# top-level project are NOT visible) and once in the real build.  Deriving
# SOUI_ROOT from this file's own location keeps both passes consistent.
#   <soui4>/soui-freertos/esp32s3/components/soui_cairo/../../../.. == <soui4>
get_filename_component(SOUI_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../.." ABSOLUTE)
if(NOT EXISTS "${SOUI_ROOT}/SOUI/include/souistd.h")
    message(FATAL_ERROR "[soui_cairo] SOUI_ROOT 推导失败: ${SOUI_ROOT}")
endif()

set(SWINX_DIR  "${SOUI_ROOT}/swinx")
set(PIXMAN_DIR "${SWINX_DIR}/thirdparty/pixman/pixman")
set(CAIRO_DIR  "${SWINX_DIR}/thirdparty/cairo/src")
set(PNG_DIR    "${SWINX_DIR}/thirdparty/libpng")
set(ZLIB_DIR   "${SWINX_DIR}/thirdparty/zlib")
set(GEN_DIR    "${CMAKE_CURRENT_BINARY_DIR}/gen")
file(MAKE_DIRECTORY "${GEN_DIR}")

# ---- pixman-version.h -------------------------------------------------------
set(PIXMAN_VERSION_MAJOR 0)
set(PIXMAN_VERSION_MINOR 46)
set(PIXMAN_VERSION_MICRO 5)
configure_file("${PIXMAN_DIR}/pixman-version.h.in"
               "${GEN_DIR}/pixman-version.h" @ONLY)

# ---- cairo's config.h (32-bit Xtensa) --------------------------------------
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    set(_soui_ptr_size 8)
else()
    set(_soui_ptr_size 4)
endif()
file(WRITE "${GEN_DIR}/config.h" "\
/* config.h -- generated by soui_cairo/CMakeLists.txt for vendored cairo */
#define CAIRO_NO_MUTEX 1
#define HAVE_GCC_LEGACY_ATOMICS 1
#define HAS_PIXMAN_GLYPHS 1
#define HAVE_STDINT_H 1
#define HAVE_INTTYPES_H 1
#define HAVE_STDLIB_H 1
#define HAVE_STRING_H 1
#define HAVE_TIME_H 1
#define HAVE_SETJMP_H 1
#define HAVE_UINT64_T 1
#define SIZEOF_INT 4
#define SIZEOF_LONG 4
#define SIZEOF_LONG_LONG 8
#define SIZEOF_SIZE_T ${_soui_ptr_size}
#define SIZEOF_VOID_P ${_soui_ptr_size}
#define PACKAGE_VERSION \"1.18.4\"
")

# ---- pixman (single-threaded: no TLS) --------------------------------------
file(GLOB PIXMAN_SRCS "${PIXMAN_DIR}/pixman*.c")
# pixman-region.c is a template #included by pixman-region16/32/64f.c;
# the SIMD files target x86/ARM/MIPS/RISC-V, none of which is Xtensa.
# pixman-arm-neon.c / pixman-arm-simd.c matter: they DO match the pixman*.c
# glob and, unlike the SSE/MMX ones, compile down to plain C (the NEON/ARM
# asm lives in .S files that are not globbed) -- so they would be built for
# Xtensa and fail on their ARM-only pixel paths.
foreach(_f pixman-mmx.c pixman-sse2.c pixman-ssse3.c pixman-vmx.c
           pixman-rvv.c pixman-mips-dspr2.c pixman-region.c
           pixman-arm-neon.c pixman-arm-simd.c)
    list(REMOVE_ITEM PIXMAN_SRCS "${PIXMAN_DIR}/${_f}")
endforeach()

# ---- zlib / libpng ----------------------------------------------------------
file(GLOB ZLIB_SRCS "${ZLIB_DIR}/*.c")
foreach(_f example.c minigzip.c)
    list(REMOVE_ITEM ZLIB_SRCS "${ZLIB_DIR}/${_f}")
endforeach()

file(GLOB PNG_SRCS "${PNG_DIR}/*.c")
foreach(_f pngtest.c pngexample.c)
    list(REMOVE_ITEM PNG_SRCS "${PNG_DIR}/${_f}")
endforeach()

# ---- cairo (image surface + png only) --------------------------------------
file(GLOB CAIRO_SRCS "${CAIRO_DIR}/*.c")
set(CAIRO_EXCLUDE
    cairo-ft-font.c cairo-quartz-font.c
    cairo-quartz-image-surface.c cairo-quartz-surface.c
    cairo-pdf-interchange.c cairo-pdf-operators.c cairo-pdf-shading.c cairo-pdf-surface.c
    cairo-ps-surface.c cairo-svg-surface.c cairo-script-surface.c
    cairo-xcb-connection-core.c cairo-xcb-connection-render.c cairo-xcb-connection-shm.c
    cairo-xcb-connection.c cairo-xcb-resources.c cairo-xcb-screen.c cairo-xcb-shm.c
    cairo-xcb-surface-core.c cairo-xcb-surface-render.c cairo-xcb-surface.c
    cairo-xlib-core-compositor.c cairo-xlib-display.c cairo-xlib-fallback-compositor.c
    cairo-xlib-render-compositor.c cairo-xlib-screen.c cairo-xlib-source.c
    cairo-xlib-surface-shm.c cairo-xlib-surface.c cairo-xlib-visual.c cairo-xlib-xcb-surface.c
    cairo-image-mask-compositor.c
    cairo-svg-glyph-render.c cairo-colr-glyph-render.c
    cairo-cff-subset.c cairo-truetype-subset.c cairo-type1-fallback.c cairo-type1-subset.c
    cairo-type1-glyph-names.c cairo-type3-glyph-surface.c cairo-scaled-font-subsets.c
    cairo-paginated-surface.c cairo-deflate-stream.c
    cairo-tee-surface.c
    test-base-compositor-surface.c test-compositor-surface.c
    test-null-compositor-surface.c test-paginated-surface.c)
# NOTE: use absolute paths here. SouiFreeRTOS.cmake removes the bare names,
# which does not match an absolute-path GLOB result (soui_fr/CMakeLists.txt
# gets this right); kept correct in this port.
foreach(_f ${CAIRO_EXCLUDE})
    list(REMOVE_ITEM CAIRO_SRCS "${CAIRO_DIR}/${_f}")
endforeach()

idf_component_register(
    SRCS ${PIXMAN_SRCS} ${ZLIB_SRCS} ${PNG_SRCS} ${CAIRO_SRCS}
    INCLUDE_DIRS
        "${CAIRO_DIR}"
        "${PIXMAN_DIR}"
        "${PNG_DIR}"
        "${ZLIB_DIR}"
    PRIV_INCLUDE_DIRS
        # cairo's config.h + pixman-version.h must win for these *.c files
        # only; exporting GEN_DIR would collide with soui's own config.h.
        "${GEN_DIR}"
        # bare-metal cairo-features.h stub
        "${SWINX_DIR}/src/platform/freertos"
)

target_compile_definitions(${COMPONENT_LIB} PRIVATE
    PIXMAN_NO_TLS
    PACKAGE=pixman)
# NOT -Os: this list lands *after* CMAKE_C_FLAGS on the compiler command line, so
# a -Os here silently wins over the project-wide -O2 that
# CONFIG_COMPILER_OPTIMIZATION_PERF puts there.  That left cairo (107 files) and
# pixman (32 files) -- i.e. the code that actually touches pixels -- built for
# size while the rest of the firmware was built for speed.
target_compile_options(${COMPONENT_LIB} PRIVATE -w -O2)

# ---------------------------------------------------------------------------
# Vendored-C warning policy under Xtensa GCC 14
# ---------------------------------------------------------------------------
# `-w` silences diagnostics but NOT the two categories that GCC 14 promoted
# from warning to *error* in C mode.  Both fire on this tree for a reason that
# is harmless on a 32-bit Xtensa target:
#
#  * -Wincompatible-pointer-types (309 sites, all in pixman-inlines.h):
#    pixman assumes pixman_fixed_t == int and passes pixman_fixed_t* where an
#    int* is expected.  On Xtensa newlib int32_t is `long int`, so the two
#    pointee types differ by name while both being 32-bit -- the generated
#    code is identical to the x86/ARM builds.
#  * -Wimplicit-function-declaration (zlib's gzguts.h): HAVE_UNISTD_H is not
#    defined for this platform, so lseek/read/write/close are used undeclared.
#    All four return 32-bit values here and resolve to newlib at link time.
#
# Keep them as warnings (visible in the log) instead of turning them into
# compile failures; do NOT blanket-disable -Werror=all.
target_compile_options(${COMPONENT_LIB} PRIVATE
    -Wno-error=incompatible-pointer-types
    -Wno-error=implicit-function-declaration
    -Wno-error=int-conversion
    -Wno-error=return-mismatch
    -Wno-error=implicit-int
    -Wno-error=declaration-missing-parameter-type)
